### [CODE 6]
def time_formatter(x, pos):
    is_neg = x < 0
    abs_x = abs(x)
    if abs_x >= 60000:
        mins, secs = int(abs_x // 60000), int((abs_x % 60000) // 1000)
        res = f'{mins}m' + (f' {secs}s' if secs > 0 else '')
    elif abs_x >= 1000:
        res = f'{abs_x/1000:g}s'
    else:
        res = f'{abs_x:g}ms'
    return f'-{res}' if is_neg and abs_x != 0 else res

def duration_formatter(d_ms):
  if d_ms < 1000:
    return f"{d_ms:.0f}ms"
  return f"{d_ms / 1000:.1f}s"

### [MD 10]
Timings:

- t0: peer finishes reconstruction, sends announcement of high-bandwidth cmpctblock to us
- t1: we receive announcement of cmpctblock (~0.5 RTT after t0)
- t2: we send getblocktxn request (assumed to be instant)
- t3: peer receives getblocktxn request (~0.5 RTT after t2)
- t4: peer finishes validation and sends blocktxn (variable)
- t5: we receive blocktxn (~0.5 RTT after t4)

We want to measure the time between t0 and t4 (d). We have t1 (and t2) and t5 and the RTT between us and the peer. However, since we have to do one full round-trip between t0 and t4, when the block validates faster than our RTT to the peer, we can only get an upper-bound on validation time. In this case, d ≈ RTT.

```
t0 = t1 - (0.5 RTT)
t2 = t1
t3 = t1 + (0.5 RTT)
t4 = t5 - (0.5 RTT)

d = t4 - t0
longer_than_rtt = d > RTT
```



### [CODE 11]
def calc_timings(t1, t5, rtt):
    t0 = t1 - (rtt / 2)
    t3 = t1 + (rtt / 2)
    t4 = t5 - (rtt / 2)

    d = t4 - t0
    longer_than_rtt = d > rtt

    return t0, t3, t4, d, longer_than_rtt

### [CODE 19]
median_rtts = df_master.groupby('peer_addr')['rtt_ms'].median().to_dict()

# State per peer: tracks the open window
peer_state = {}  # peer_addr -> {block_hash, t_start, connection_id, run}
results = []

for row in df_master.itertuples():
    peer_addr = row.peer_addr
    state = peer_state.get(peer_addr)

    # Reset state on connection change
    if state and state['connection_id'] != row.connection_id:
        state = None
        peer_state.pop(peer_addr, None)

    # --- Try to close an open window first ---
    if state is not None:
        if row.announcement_type == 5 and row.block_hash == state['block_hash']:
            # Completed with a BlockTxn for the same block
            seq_type = '4-to-5'
        elif row.announcement_type in [3, 4] and row.block_hash != state['block_hash']:
            # New block arrived before we got a type 5 → close as timeout
            seq_type = '4-to-3'
        else:
            seq_type = None

        if seq_type:
            rtt = median_rtts.get(peer_addr, row.rtt_ms)
            _, _, _, d, longer_than_rtt = calc_timings(state['t_start'], row.timestamp_ms, rtt)
            results.append({
                'peer_addr': peer_addr,
                'prev_block_hash': state['block_hash'],
                'trigger_block_hash': row.block_hash,
                'trigger_type': row.announcement_type,
                'type': seq_type,
                'duration_ms': d,
                'run': state['run'],
                'median_rtt_ms': rtt,
                'network': row.network,
                'longer_than_rtt': longer_than_rtt,
            })
            # Window is closed
            state = None
            peer_state.pop(peer_addr, None)

    # --- Open a new window on type 4 (only if no window is currently open) ---
    if row.announcement_type == 4 and state is None:
        peer_state[peer_addr] = {
            'block_hash': row.block_hash,
            't_start': row.timestamp_ms,
            'connection_id': row.connection_id,
            'run': row.run,
        }

df_results = pd.DataFrame(results)
df_results['network'] = df_results['peer_addr'].apply(
    lambda x: 'onion' if '.onion' in x else 'ipv4'
)

display(df_results.head())
print(f'Total sequences captured: {len(df_results)}')
